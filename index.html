cursor.execute(""" CREATE TABLE IF NOT EXISTS users ( user_id INTEGER PRIMARY KEY, username TEXT, coins INTEGER DEFAULT 100, xp INTEGER DEFAULT 0, level INTEGER DEFAULT 1, guild_id INTEGER, referrer_id INTEGER, last_daily INTEGER DEFAULT 0 ) """) cursor.execute(""" CREATE TABLE IF NOT EXISTS guilds ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT UNIQUE, owner_id INTEGER, score INTEGER conn.commit() conn.close() def db_execute(query, params=(), fetchone=False, fetchall=False, commit=False): conn = sqlite3.connect(DB_NAME) cursor = conn.cursor() cursor.execute(query, params) res = None if commit: conn.commit() if fetchone: res = cursor.fetchone() if fetchall: res = cursor.fetchall() conn.close() return res # --- СОСТОЯНИЯ FSM --- class QuestStates(StatesGroup): title = State()
